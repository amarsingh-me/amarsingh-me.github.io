{{- /* Resolves Obsidian-vault-style relative links (e.g. [Text](../folder/note.md)) to the note's real published URL, so the same link works both in Obsidian and on the built site. Anything else (http/https/mailto/anchors/absolute Hugo paths) falls through to the theme's default link rendering. */ -}} {{- $dest := .Destination -}} {{- $isNoteLink := and (strings.HasSuffix $dest ".md") (not (strings.HasPrefix $dest "http://")) (not (strings.HasPrefix $dest "https://")) -}} {{- if $isNoteLink -}} {{- with .Page.GetPage $dest -}} {{- $.Text | safeHTML -}} {{- else -}} {{- $.Text | safeHTML -}} {{- end -}} {{- else -}} {{- .Text | safeHTML -}} {{- end -}} {{- /* Trim EOF */ -}}
